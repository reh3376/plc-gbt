## 🚨 CRITICAL: Zero-Tolerance Policy for Manual API Definitions

**MANDATORY RULE**: All API definitions, JSON schemas, and UI schemas MUST use the OpenAPI Schema MCP from MCP_Docker server. **NO EXCEPTIONS**.

### ⚠️ **NEVER Manually Define API Schemas**

**CRITICAL ERROR PATTERN TO AVOID:**
```typescript
// ❌ WRONG - Manual API definitions cause immediate build failures
interface UserAPI {
  id: string
  name: string
  // Manual definitions = Schema drift = Production failures
}

// ❌ WRONG - Manual Zod schemas bypass OpenAPI governance
const UserSchema = z.object({
  id: z.string(),
  name: z.string()
})
```

**✅ CORRECT APPROACH:**
```typescript
// ✅ RIGHT - All schemas from OpenAPI Schema MCP
import { openAPISchemaMCP } from '@/lib/mcp/openapi-schema-client'
import type { components } from '@/api/types.gen' // Generated from OpenAPI

// Types are generated, never manually written
type User = components['schemas']['User']

// Zod schemas are generated from OpenAPI
import { UserSchema } from '@/api/zod-schemas' // Generated by openapi-zod-client
```

## 🎯 Core Principles

1. **OpenAPI Schema MCP is the Single Source of Truth** - All API contracts MUST be defined using the OpenAPI Schema MCP tool
2. **Zero Manual Type Definitions** - All types are generated from OpenAPI schemas
3. **Runtime Validation Everywhere** - Every API request/response must be validated with Zod schemas
4. **Strict TypeScript Compliance** - Zero `any` types policy throughout the codebase
5. **CI/CD Enforcement** - Build fails if schemas are out of sync
6. **Two-Phase Testing** - >95% automated testing + mandatory user validation before deployment

#### Type-Safe API Methods
```typescript
// src/lib/api/workflows.ts
import { apiClient } from './client';
import { WorkflowDataSchema, WorkflowCreateSchema } from '@/api/zod-schemas';
import type { WorkflowData, WorkflowCreate } from '@/api/types.gen';

export const workflowsApi = {
  // GET with response validation
  async getById(id: string): Promise<WorkflowData> {
    const { data } = await apiClient.get(`/v1/workflows/${id}`);
    
    // MANDATORY: Validate response against generated schema
    const result = WorkflowDataSchema.safeParse(data.data);
    if (!result.success) {
      throw new ValidationError('Invalid workflow data from API', result.error);
    }
    
    return result.data;
  },

  // POST with request & response validation
  async create(input: unknown): Promise<WorkflowData> {
    // Validate input against generated schema
    const payload = WorkflowCreateSchema.parse(input);
    
    const { data } = await apiClient.post('/v1/workflows', payload);
    
    // Validate response
    const result = WorkflowDataSchema.safeParse(data.data);
    if (!result.success) {
      throw new ValidationError('Invalid response from create API', result.error);
    }
    
    return result.data;
  },

  // PUT with optimistic updates
  async update(id: string, input: unknown): Promise<WorkflowData> {
    const payload = WorkflowUpdateSchema.parse(input);
    
    const { data } = await apiClient.put(`/v1/workflows/${id}`, payload);
    
    return WorkflowDataSchema.parse(data.data);
  }
```

```typescript
// src/features/workflows/components/WorkflowForm.tsx
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import { WorkflowCreateSchema } from '@/api/zod-schemas';

// Extend API schema with UI-specific validations
const WorkflowFormSchema = WorkflowCreateSchema.extend({
  // UI-only: Transform comma-separated tags to array
  tags: z.string().transform((val) => val.split(',').map(tag => tag.trim())),
  // UI-only: Coerce string dates
  scheduledDate: z.coerce.date().optional(),
});

type WorkflowFormData = z.infer<typeof WorkflowFormSchema>;

export function WorkflowForm() {
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
    setError,
  } = useForm<WorkflowFormData>({
    resolver: zodResolver(WorkflowFormSchema),
    defaultValues: {
      name: '',
      description: '',
      tags: '',
      priority: 'medium',
    },
  });

  const onSubmit = async (formData: WorkflowFormData) => {
    try {
      // API call with automatic validation
      const workflow = await workflowsApi.create(formData);
      toast.success('Workflow created successfully');
      router.push(`/workflows/${workflow.id}`);
    } catch (error) {
      if (error instanceof ValidationError) {
        // Map API validation errors to form fields
        error.issues.forEach((issue) => {
          const field = issue.path[0] as keyof WorkflowFormData;
          setError(field, { message: issue.message });
        });
      } else {
        toast.error('Failed to create workflow');
      }
    }
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register('name')} />
      {errors.name && <span>{errors.name.message}</span>}
      
      <textarea {...register('description')} />
      {errors.description && <span>{errors.description.message}</span>}
      
      <button type="submit" disabled={isSubmitting}>
        {isSubmitting ? 'Creating...' : 'Create Workflow'}
      </button>
    </form>
  );
}
```

## 🚨 Fool-Proof Error Handling Patterns

### Centralized Error Handling

**MANDATORY**: All API errors must be normalized and typed.

```typescript
// src/lib/api/errors.ts
import { z } from 'zod';
import { AxiosError } from 'axios';

// Standard API error schema (should match backend)
export const ApiErrorSchema = z.object({
  code: z.string(),
  message: z.string(),
  details: z.record(z.unknown()).optional(),
  timestamp: z.string().datetime(),
  path: z.string(),
});

export type ApiError = z.infer<typeof ApiErrorSchema>;

// Custom error classes
export class ValidationError extends Error {
  constructor(
    message: string,
    public readonly zodError: z.ZodError
  ) {
    super(message);
    this.name = 'ValidationError';
  }

  get issues() {
    return this.zodError.issues;
  }
}

export class ApiRequestError extends Error {
  constructor(
    message: string,
    public readonly status: number,
    public readonly apiError?: ApiError
  ) {
    super(message);
    this.name = 'ApiRequestError';
  }
}

// Normalize any error to ApiError format
export function normalizeApiError(error: unknown): ApiError {
  // Handle Axios errors
  if (error instanceof AxiosError) {
    const data = error.response?.data;
    
    // Try to parse as standard API error
    const parsed = ApiErrorSchema.safeParse(data);
    if (parsed.success) {
      return parsed.data;
    }
    
    // Fallback for non-standard errors
    return {
      code: error.code || 'UNKNOWN',
      message: error.message || 'An unexpected error occurred',
      details: { response: data },
      timestamp: new Date().toISOString(),
      path: error.config?.url || 'unknown',
    };
  }
  
  // Handle validation errors
  if (error instanceof z.ZodError) {
    return {
      code: 'VALIDATION_ERROR',
      message: 'Validation failed',
      details: { issues: error.issues },
      timestamp: new Date().toISOString(),
      path: 'client-validation',
    };
  }
  
  // Generic error fallback
  return {
    code: 'UNKNOWN_ERROR',
    message: error instanceof Error ? error.message : 'Unknown error',
    details: {},
    timestamp: new Date().toISOString(),
    path: 'unknown',
  };
}
```

### 2. Schema Definition Rules

- **Always define in OpenAPI first** - Never create types manually
- **Use $ref for reusability** - Reference common schemas
- **Include required fields** - Be explicit about mandatory properties
- **Add descriptions** - Document what each field represents

### 3. API Route Requirements

Every API route MUST:
1. Import `openAPISchemaMCP`
2. Define clear request/response types
3. Validate all inputs with Zod
4. Validate all outputs with `openAPISchemaMCP.validateResponse()`
5. Handle validation failures gracefully
6. Return consistent error responses

### 4. Mock Data Generation

When creating mock data, ensure it conforms to schemas:

## ✅ Do's and Don'ts

### Do's
- ✅ **Always** use OpenAPI Schema MCP for defining API contracts
- ✅ **Always** validate responses before sending
- ✅ **Always** validate requests before processing
- ✅ **Always** handle validation errors explicitly
- ✅ **Always** keep schemas in sync with types
- ✅ Use descriptive schema names and properties
- ✅ Version your APIs properly (`/v1/`, `/v2/`)
- ✅ Test schema validation in unit tests
- ✅ Use `safeParse` for recoverable errors
- ✅ Use `parse` when you want to throw
- ✅ Implement retry logic for transient failures
- ✅ Add request/response logging in development

### Don'ts
- ❌ **Never** manually define API types
- ❌ **Never** skip validation "for simplicity"
- ❌ **Never** use `any` type in API definitions
- ❌ **Never** modify generated files
- ❌ **Never** trust unvalidated JSON data
- ❌ Don't mix UI-specific validation with API validation
- ❌ Don't create APIs without OpenAPI documentation
- ❌ Don't catch errors without proper handling
- ❌ Don't use `as` type assertions on API data
- ❌ Don't forget to handle loading and error states

## ⚠️ Common Pitfalls & How to Avoid Them

### 1. Schema Drift
**Problem**: Frontend types don't match backend API
**Solution**: 
```bash
# Add to CI/CD pipeline
pnpm api:pull && pnpm api:gen
git diff --exit-code || (echo "Schema drift detected!" && exit 1)
```

### 2. Optional vs Required Fields
**Problem**: Confusion about which fields are optional
**Solution**: Be explicit in OpenAPI schema
```typescript
// OpenAPI Schema
properties: {
  id: { type: 'string' },           // Required (in required array)
  description: { type: 'string' },   // Optional (not in required array)
}
required: ['id']  // Only id is required
```

### 3. Nested Object Validation
**Problem**: Deeply nested objects fail validation
**Solution**: Use proper $ref definitions
```typescript
// Define nested schemas separately
PersonName: {
  type: 'object',
  properties: {
    first: { type: 'string' },
    last: { type: 'string' }
  },
  required: ['first', 'last']
}

Person: {
  type: 'object',
  properties: {
    name: { $ref: '#/components/schemas/PersonName' }
  }
}
```

### 4. Array Validation Issues
**Problem**: Empty arrays or wrong item types
**Solution**: Specify array constraints
```typescript
tags: {
  type: 'array',
  items: { type: 'string' },
  minItems: 0,
  maxItems: 10,
  uniqueItems: true
}
```

### Scenario: Creating a Task Management API

#### 1. Define OpenAPI Schema
```typescript
// openapi-schema-client.ts
export const schemas = {
  Task: {
    type: 'object',
    required: ['id', 'title', 'status', 'createdAt'],
    properties: {
      id: { type: 'string', format: 'uuid' },
      title: { type: 'string', minLength: 1, maxLength: 200 },
      description: { type: 'string', maxLength: 1000 },
      status: { enum: ['pending', 'in_progress', 'completed', 'cancelled'] },
      priority: { enum: ['low', 'medium', 'high'], default: 'medium' },
      dueDate: { type: 'string', format: 'date-time' },
      createdAt: { type: 'string', format: 'date-time' },
      updatedAt: { type: 'string', format: 'date-time' }
    }
  },
  
  TaskCreate: {
    type: 'object',
    required: ['title'],
    properties: {
      title: { type: 'string', minLength: 1, maxLength: 200 },
      description: { type: 'string', maxLength: 1000 },
      priority: { enum: ['low', 'medium', 'high'], default: 'medium' },
      dueDate: { type: 'string', format: 'date-time' }
    }
  }
};
```

#### 2. Implement API Routes
```typescript
// app/api/v1/tasks/route.ts
export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    
    // Validate request
    const createData = TaskCreateSchema.parse(body);
    
    // Create task
    const newTask: Task = {
      id: generateUUID(),
      ...createData,
      status: 'pending',
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };
    
    // Save to database
    await db.tasks.create(newTask);
    
    // Prepare response
    const response = {
      success: true,
      message: 'Task created successfully',
      data: newTask
    };
    
    // Validate response
    await openAPISchemaMCP.validateResponse('POST', '/api/v1/tasks', 201, response);
    
    return NextResponse.json(response, { status: 201 });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { code: 'VALIDATION_ERROR', message: 'Invalid input', details: error.issues },
        { status: 400 }
      );
    }
    
    return NextResponse.json(
      { code: 'INTERNAL_ERROR', message: 'Failed to create task' },
      { status: 500 }
    );
  }
}
```

#### 3. Create Type-Safe API Client
```typescript
// lib/api/tasks.ts
export const tasksApi = {
  async create(input: unknown): Promise<Task> {
    const payload = TaskCreateSchema.parse(input);
    const { data } = await apiClient.post('/v1/tasks', payload);
    return TaskSchema.parse(data.data);
  },
  
  async list(params?: { status?: TaskStatus; page?: number }): Promise<PaginatedTasks> {
    const { data } = await apiClient.get('/v1/tasks', { params });
    return PaginatedTasksSchema.parse(data);
  },
  
  async update(id: string, updates: unknown): Promise<Task> {
    const payload = TaskUpdateSchema.parse(updates);
    const { data } = await apiClient.patch(`/v1/tasks/${id}`, payload);
    return TaskSchema.parse(data.data);
  }
};
```
